cmake_minimum_required(VERSION 3.21)
project(DesktopPet VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(DESKTOPPET_OWNER "" CACHE STRING "GitHub owner")
set(DESKTOPPET_REPOSITORY "" CACHE STRING "GitHub repository")
set(DESKTOPPET_PUBLIC_KEY "" CACHE STRING "Trusted ECDSA P256 X||Y, 128 hex characters")
foreach(v DESKTOPPET_OWNER DESKTOPPET_REPOSITORY)
  if(NOT "${${v}}" MATCHES "^[A-Za-z0-9_.-]*$")
    message(FATAL_ERROR "Invalid ${v}")
  endif()
endforeach()
if(NOT DESKTOPPET_PUBLIC_KEY STREQUAL "")
  string(LENGTH "${DESKTOPPET_PUBLIC_KEY}" key_length)
  if(NOT key_length EQUAL 128 OR NOT DESKTOPPET_PUBLIC_KEY MATCHES "^[A-Fa-f0-9]+$")
    message(FATAL_ERROR "Public key must be a P256 X||Y point")
  endif()
endif()
configure_file(src/build_config.h.in generated/build_config.h @ONLY)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive- /EHsc)
endif()
add_library(petcore INTERFACE)
target_include_directories(petcore INTERFACE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE petcore)
add_test(NAME core COMMAND core_tests)
if(WIN32)
  enable_language(RC)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  add_library(petwin STATIC src/platform.cpp src/update.cpp)
  target_link_libraries(petwin PUBLIC petcore bcrypt winhttp shell32 ole32 advapi32 user32 gdi32)
  add_executable(DesktopPet WIN32 src/main.cpp src/pet.rc)
  target_link_libraries(DesktopPet PRIVATE petwin comctl32)
  target_include_directories(DesktopPet PRIVATE assets/icons)
  add_executable(Updater WIN32 updater/main.cpp src/pet.rc)
  target_include_directories(Updater PRIVATE assets/icons)
  target_link_libraries(Updater PRIVATE petwin)
  add_executable(windows_tests tests/windows_tests.cpp)
  target_link_libraries(windows_tests PRIVATE petwin)
  add_test(NAME windows_native COMMAND windows_tests)
  if(MSVC)
    target_link_options(DesktopPet PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest)
    target_link_options(Updater PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest)
  endif()
  install(TARGETS DesktopPet Updater RUNTIME DESTINATION .)
  install(FILES data/dialogue.json DESTINATION data)
  install(FILES README.md CHANGELOG.md DESTINATION .)
endif()
