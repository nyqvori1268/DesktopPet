cmake_minimum_required(VERSION 3.21)
project(DesktopPet VERSION 1.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(DESKTOPPET_OWNER "nyqvori1268" CACHE STRING "GitHub owner")
set(DESKTOPPET_REPOSITORY "DesktopPet" CACHE STRING "GitHub repository")
set(DESKTOPPET_PUBLIC_KEY "" CACHE STRING "Trusted ECDSA P256 X||Y, 128 hex characters")
set(DESKTOPPET_BACKUP_PUBLIC_KEY "" CACHE STRING "Offline backup trust anchor, P256 X||Y")
set(DESKTOPPET_DEVELOPER "Meow" CACHE STRING "Developer display name")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" license_contents)
string(REGEX MATCH "Copyright [^\r\n]+" license_copyright "${license_contents}")
string(REPLACE "Copyright (c) " "© " license_copyright "${license_copyright}")
set(DESKTOPPET_COPYRIGHT "${license_copyright}" CACHE STRING "Copyright display notice")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE" license_text)
foreach(meta DESKTOPPET_DEVELOPER DESKTOPPET_COPYRIGHT license_text)
  string(REPLACE "\\" "\\\\" ${meta}_ESCAPED "${${meta}}")
  string(REPLACE "\"" "\\\"" ${meta}_ESCAPED "${${meta}_ESCAPED}")
  string(REPLACE "\n" "\\n" ${meta}_ESCAPED "${${meta}_ESCAPED}")
  string(REPLACE "\r" "" ${meta}_ESCAPED "${${meta}_ESCAPED}")
endforeach()
foreach(v DESKTOPPET_OWNER DESKTOPPET_REPOSITORY)
  if(NOT "${${v}}" MATCHES "^[A-Za-z0-9_.-]*$")
    message(FATAL_ERROR "Invalid ${v}")
  endif()
endforeach()
if(NOT DESKTOPPET_PUBLIC_KEY STREQUAL "")
  string(LENGTH "${DESKTOPPET_PUBLIC_KEY}" key_length)
  if(NOT key_length EQUAL 128 OR NOT DESKTOPPET_PUBLIC_KEY MATCHES "^[A-Fa-f0-9]+$")
    message(FATAL_ERROR "Public key must be a P256 X||Y point")
  endif()
endif()
if(NOT DESKTOPPET_BACKUP_PUBLIC_KEY STREQUAL "")
  string(LENGTH "${DESKTOPPET_BACKUP_PUBLIC_KEY}" backup_length)
  if(NOT backup_length EQUAL 128 OR NOT DESKTOPPET_BACKUP_PUBLIC_KEY MATCHES "^[A-Fa-f0-9]+$")
    message(FATAL_ERROR "Backup key must be a P256 X||Y point")
  endif()
  string(TOLOWER "${DESKTOPPET_PUBLIC_KEY}" primary_point)
  string(TOLOWER "${DESKTOPPET_BACKUP_PUBLIC_KEY}" backup_point)
  if(primary_point STREQUAL backup_point)
    message(FATAL_ERROR "Primary and backup keys must differ")
  endif()
endif()
configure_file(src/build_config.h.in generated/build_config.h @ONLY)
configure_file(src/pet.rc.in generated/pet.rc @ONLY)
configure_file(src/ocean.rc.in generated/ocean.rc @ONLY)
configure_file(src/renewed.rc.in generated/renewed.rc @ONLY)
configure_file(src/app.manifest.in generated/app.manifest @ONLY)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive- /EHsc $<$<COMPILE_LANGUAGE:CXX>:/GS> $<$<COMPILE_LANGUAGE:CXX>:/guard:cf>)
  add_link_options(/DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT /GUARD:CF /DEPENDENTLOADFLAG:0x800 /INCREMENTAL:NO)
endif()
add_library(petcore INTERFACE)
target_include_directories(petcore INTERFACE src "${CMAKE_CURRENT_BINARY_DIR}/generated")
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE petcore)
target_compile_definitions(core_tests PRIVATE PET_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
add_test(NAME core COMMAND core_tests)
add_executable(renewed_tests tests/renewed_tests.cpp)
target_link_libraries(renewed_tests PRIVATE petcore)
target_compile_definitions(renewed_tests PRIVATE PET_TEST_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME renewed_art COMMAND renewed_tests)
add_executable(security_tests tests/security_tests.cpp)
target_link_libraries(security_tests PRIVATE petcore)
add_test(NAME security_recovery COMMAND security_tests)
if(WIN32)
  enable_language(RC)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  add_library(petwin STATIC src/platform.cpp src/update.cpp src/handoff.cpp src/trust_store.cpp src/transaction.cpp)
  target_link_libraries(petwin PUBLIC petcore bcrypt crypt32 winhttp shell32 ole32 advapi32 user32 gdi32)
  add_executable(DesktopPet WIN32 src/main.cpp src/custom_pet.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/pet.rc" "${CMAKE_CURRENT_BINARY_DIR}/generated/ocean.rc" "${CMAKE_CURRENT_BINARY_DIR}/generated/renewed.rc")
  target_link_libraries(DesktopPet PRIVATE petwin comctl32 gdiplus comdlg32)
  target_include_directories(DesktopPet PRIVATE assets/icons)
  add_executable(Updater WIN32 updater/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/generated/pet.rc")
  target_include_directories(Updater PRIVATE assets/icons)
  target_link_libraries(Updater PRIVATE petwin)
  add_executable(windows_tests tests/windows_tests.cpp src/custom_pet.cpp)
  target_compile_definitions(windows_tests PRIVATE PET_TEST_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  target_link_libraries(windows_tests PRIVATE petwin gdiplus)
  add_library(dll_probe SHARED tests/dll_probe.cpp)
  add_test(NAME dll_search_policy COMMAND windows_tests --dll-policy $<TARGET_FILE:dll_probe>)
  add_test(NAME windows_native COMMAND windows_tests)
  add_test(NAME ocean_embedded COMMAND windows_tests --ocean-resources $<TARGET_FILE:DesktopPet>)
  add_test(NAME renewed_embedded COMMAND windows_tests --renewed-resources $<TARGET_FILE:DesktopPet>)
  if(MSVC)
    target_link_options(DesktopPet PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/generated/app.manifest)
    target_link_options(Updater PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/generated/app.manifest)
  endif()
  install(TARGETS DesktopPet Updater RUNTIME DESTINATION .)
  install(FILES data/dialogue.json DESTINATION data)
  install(FILES README.md CHANGELOG.md LICENSE THIRD_PARTY_NOTICES.md DESTINATION .)
endif()
